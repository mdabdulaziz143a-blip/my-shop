অনলাইন অর্ডার ফর্ম

📋 নির্দেশিকা:
  1. নিচের ফর্মটি পূরণ করুন
  2. সব তথ্য সঠিকভাবে দিন
  3. সাবমিট করলে অর্ডারটি আমাদের ডাটাবেসে চলে যাবে
  4. আমরা আপনাকে কল করে কনফার্ম করব
✅ ফর্ম সফলভাবে সাবমিট করার পর:
  • "ধন্যবাদ" মেসেজ দেখবেন
  • ডাটা সরাসরি Google Sheet-এ সেভ হবে
  • আপনি Admin Panel থেকে সব অর্ডার দেখতে পারবেন
  • আমরা ১৫-২০ মিনিটের মধ্যে আপনার সাথে যোগাযোগ করব

👑 এডমিন কন্ট্রোল প্যানেল

এই সেকশন শুধুমাত্র দোকানের Owner/Admin এর জন্য। এখান থেকে সব অর্ডার দেখতে和管理 করতে পারবেন।

📊 সব অর্ডার দেখুন ✏️ ফর্ম এডিট করুন

🚀 মেইল নোটিফিকেশন সেটআপ করুন (Step-by-Step)

প্রতিটি অর্ডার পেলে মেইল পেতে এই স্টেপগুলো ফলো করুন:

📌 ধাপ ১: উপরের "সব অর্ডার দেখুন" বাটনে ক্লিক করে Google Sheets ওপেন করুন

📌 ধাপ ২: Sheets মেনু থেকে Extensions → Apps Script এ ক্লিক করুন

📌 ধাপ ৩: নিচের কোডটি কপি করে Apps Script এ পেস্ট করুন:

function onFormSubmit(e) {
  var সময় = e.values[0];
  var নাম = e.values[1];
  var ফোন = e.values[2];
  var ঠিকানা = e.values[3];
  var পণ্য = e.values[4];
  var মোট_দাম = e.values[5];
  var অবস্থা = e.values[6];

  var আপনার_ইমেইল = "আপনার-ইমেইল-এখানে-লিখুন@gmail.com";

  var মেইল_শিরোনাম = "🛒 নতুন অর্ডার! - " + নাম;
  var মেইল_বিস্তারিত = "আরিফা মুদি ভান্ডারে নতুন অর্ডার!\\n\\n" +
  "গ্রাহক: " + নাম + "\\n" +
  "ফোন: " + ফোন + "\\n" +
  "ঠিকানা: " + ঠিকানা + "\\n" +
  "পণ্য: " + পণ্য + "\\n" +
  "মূল্য: " + মোট_দাম + " টাকা\\n\\n" +
  "দ্রুত যোগাযোগ করুন।";

  MailApp.sendEmail(আপনার_ইমেইল, মেইল_শিরোনাম, মেইল_বিস্তারিত);
}

📌 ধাপ ৪: লাইন ১০-এ আপনার নিজের ইমেইল ঠিকানা লিখুন

📌 ধাপ ৫: Save করুন → বামপাশে Triggers (⏰) আইকনে ক্লিক করুন → "+ Add Trigger"

📌 ধাপ ৬: ফাংশন: onFormSubmit, ইভেন্ট: On form submit সিলেক্ট করে Save করুন
